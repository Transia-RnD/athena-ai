```cpp
// OpenViewSandbox.h

class OpenViewSandbox
{
private:
    // Instead of OpenView&, use OpenView*
    OpenView* parentView_;
    std::unique_ptr<OpenView> sandbox_;

public:
    using key_type = ReadView::key_type;

    // Accepts a pointer to the parent view (can be a sandbox or the base view)
    OpenViewSandbox(OpenView* parentView)
        : parentView_(parentView)
        , sandbox_(std::make_unique<OpenView>(OpenView::batch_view, *parentView))
    {
    }

    // ... (rawErase, rawInsert, etc. unchanged)

    void commit()
    {
        sandbox_->apply(*parentView_);
        // Reset sandbox for possible reuse
        sandbox_ = std::make_unique<OpenView>(OpenView::batch_view, *parentView_);
    }

    void discard()
    {
        sandbox_ = std::make_unique<OpenView>(OpenView::batch_view, *parentView_);
    }

    OpenView const& view() const { return *sandbox_; }
    OpenView& view() { return *sandbox_; }
};

// Usage pattern for nested sandboxes:

// Top-level (root) sandbox
OpenView baseView = ...; // The base ledger view
OpenViewSandbox rootSandbox(&baseView);

// ... perform operations on rootSandbox ...

// When emitting a transaction (nested contract call):
OpenViewSandbox childSandbox(&rootSandbox.view());

// ... perform operations on childSandbox ...

// When ready, commit child changes to parent:
childSandbox.commit();

// Later, commit parent to base:
rootSandbox.commit();
```

**Key Points:**
- Each OpenViewSandbox now takes a pointer to its parent view (which can be another sandbox's view).
- When you emit a transaction, create the child sandbox with the parent sandbox's current view.
- This allows the child to see all uncommitted changes from the parent.
- Commit order is child-to-parent-to-base, preserving atomicity and rollback.

**This is a composable, layered view stack.** It is safe as long as you only commit in order (child before parent), and you do not expose the base view until all sandboxes are committed.
